}            public void onNothingSelected(AdapterView<?> parent) {}        });    }    private void addInventoryItem() {        if (saving) return;        final int id = selectedItemId;                String name = itemNameInput.getText().toString().trim();        String sku = itemSkuInput.getText().toString().trim();        String q = itemQuantityInput.getText().toString().trim();        if (!isInventoryInputValid(name,sku,q)) return;        int quantity, minimum;        try { quantity = InputRules.quantity(q); }        catch (IllegalArgumentException e) { itemQuantityInput.setError(e.getMessage()); return; }        try { minimum = InputRules.minimumStock(minimumStockInput.getText().toString().trim()); }        catch (IllegalArgumentException e) { minimumStockInput.setError(e.getMessage()); return; }        String category = categoryInput.getText().toString().trim();        String location = locationInput.getText().toString().trim();        }, result -> {            saving = false;            if (result == 1) {                Toast.makeText(this,"Inventory item saved",Toast.LENGTH_SHORT).show();                showSmsResult(name,quantity,minimum);
